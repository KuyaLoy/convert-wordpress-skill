# {{SITE_NAME}}, WordPress: the live .htaccess. Replaces the static site's .htaccess at launch.
<!-- if:redirects -->
# Old-URL redirects live in Tools > Redirection (editable); this file only keeps the host rules, caching and
# compression from the static site, the private-files block and the WordPress block.
<!-- /if:redirects -->
<!-- unless:redirects -->
# This build keeps every old URL, so it has no redirects; this file keeps the host rules, caching and compression
# from the static site, the private-files block and the WordPress block.
<!-- /unless:redirects -->
#
# If the old .htaccess had a "php -- BEGIN cPanel-generated handler" block, paste that block back in at the top.

Options -Indexes

<IfModule mod_rewrite.c>
  RewriteEngine On

  # The site's address: https://{{LIVE_HOST}}
  # {{OTHER_HOST}} goes there, and http goes to https, in ONE hop. new-site.py wrote these rules from
  # site.json "domain" and "www"; a copy of this file for another domain changes both hosts.
  RewriteCond %{REQUEST_URI} !^/\.well-known/
  RewriteCond %{HTTP_HOST} ^{{OTHER_HOST_REGEX}}$ [NC]
  RewriteRule ^ https://{{LIVE_HOST}}%{REQUEST_URI} [R=301,L]
  RewriteCond %{REQUEST_URI} !^/\.well-known/
  RewriteCond %{HTTP_HOST} ^{{LIVE_HOST_REGEX}}$ [NC]
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^ https://{{LIVE_HOST}}%{REQUEST_URI} [R=301,L]
</IfModule>

# WordPress's readme, licence and sample config show its version, and XML-RPC is a login door the site never uses:
# none of them is public.
<FilesMatch "^(readme\.html|license\.txt|wp-config-sample\.php|xmlrpc\.php)$">
  Require all denied
</FilesMatch>

# The theme's developer files are not part of the site, and neither are its ACF field definitions (ACF reads
# acf-json from disk; this rule closes it on live whether or not the folder's own .htaccess was packed).
<IfModule mod_alias.c>
<!-- starter:tw -->
  # _tw: WordPress loads only the theme/ folder; the build folder around it (sources, tools) stays closed.
  RedirectMatch 404 ^/wp-content/themes/{{THEME}}/(?!theme/)
<!-- /starter:tw -->
  RedirectMatch 404 ^/wp-content/themes/{{THEME}}/(.*/)?(acf-json|node_modules|vendor|tests|\.git[^/]*)(/|$)
  RedirectMatch 404 ^/wp-content/themes/{{THEME}}/(.*/)?(composer\.(json|lock)|package(-lock)?\.json|phpcs\.xml\.dist)$
  RedirectMatch 404 ^/wp-content/themes/{{THEME}}/.*\.(md|log)$
</IfModule>

# Basic security headers (they change nothing on the page).
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
</IfModule>

# Planning files never go live, but block them in case they are uploaded by mistake.
<FilesMatch "^(BUILD-MAP|handoff|ai-handoff-summary|PLAN)\.md$|^debug\.log$">
  Require all denied
</FilesMatch>
<IfModule mod_alias.c>
  RedirectMatch 403 ^/(_plan|_setup|dup-installer)(/|$)
</IfModule>

# Duplicator leftovers (the installer, its backup copy, its boot log, the archive): never reachable on live.
<FilesMatch "^installer(-backup)?\.php$|^dup-installer-bootlog__.*\.txt$|_archive\.(zip|daf)$">
  Require all denied
</FilesMatch>

# BEGIN WordPress
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
RewriteBase /
RewriteRule ^index\.php$ - [L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /index.php [L]
</IfModule>
# END WordPress

<IfModule mod_mime.c>
  AddType image/webp .webp
  AddType image/avif .avif
</IfModule>

<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType image/webp "access plus 30 days"
  ExpiresByType image/avif "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/svg+xml "access plus 30 days"
  ExpiresByType text/css "access plus 7 days"
  ExpiresByType application/javascript "access plus 7 days"
  ExpiresByType text/javascript "access plus 7 days"
  ExpiresByType font/woff2 "access plus 30 days"
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript application/javascript application/json application/xml text/xml image/svg+xml
</IfModule>
